remappingStrategy {
  wifiSecurity {
    isNone = """if(
      realValue('BeaconType') == 'None' and
      realValue('BasicEncryptionModes') == 'None' and
      realValue('BasicAuthenticationMode') == 'OpenSystem'
    ) 'NONE'"""

    setNone = """if(value == 'NONE')(
      map('BeaconType' -> 'None',
          'BasicEncryptionModes' -> 'None',
          'BasicAuthenticationMode' -> 'OpenSystem',
          'WPAEncryptionModes' -> 'TKIPEncryption',
          'WPAAuthenticationMode' -> 'PSKAuthentication',
          'IEEE11iEncryptionModes' -> 'AESEncryption',
          'IEEE11iAuthenticationMode' -> 'PSKAuthentication'
      ).get(key.keyName)
    )"""

    isWep = """if(
      realValue('BeaconType') == 'Basic' and
      realValue('BasicEncryptionModes') == 'WEPEncryption' and
      realValue('BasicAuthenticationMode') == 'SharedAuthentication'
    ) 'WEP'"""

    setWep = """if(value == 'WEP')(
      map('BeaconType' -> 'Basic',
          'BasicEncryptionModes' -> 'WEPEncryption',
          'BasicAuthenticationMode' -> 'SharedAuthentication',
          'WPAEncryptionModes' -> 'TKIPEncryption',
          'WPAAuthenticationMode' -> 'PSKAuthentication',
          'IEEE11iEncryptionModes' -> 'AESEncryption',
          'IEEE11iAuthenticationMode' -> 'PSKAuthentication'
      ).get(key.keyName)
    )"""

    isWpa = """if(
      realValue('BeaconType') == 'WPA' and
      realValue('WPAEncryptionModes') == 'TKIPEncryption' and
      realValue('WPAAuthenticationMode') == 'PSKAuthentication'
    ) 'WPA'"""

    setWpa = """if(value == 'WPA')(
      map('BeaconType' -> 'WPA',
          'WPAEncryptionModes' -> 'TKIPEncryption',
          'WPAAuthenticationMode' -> 'PSKAuthentication',
          'BasicEncryptionModes' -> 'WEPEncryption',
          'BasicAuthenticationMode' -> 'SharedAuthentication',
          'IEEE11iEncryptionModes' -> 'AESEncryption',
          'IEEE11iAuthenticationMode' -> 'PSKAuthentication'
      ).get(key.keyName)
    )"""

    isWpa2 = """if(
      realValue('BeaconType') == '11i' and
      realValue('IEEE11iEncryptionModes') == 'AESEncryption' and
      realValue('IEEE11iAuthenticationMode') == 'PSKAuthentication'
    ) 'WPA2'"""

    setWpa2 = """if(value == 'WPA2')(
      map('BeaconType' -> '11i',
          'IEEE11iEncryptionModes' -> 'AESEncryption',
          'IEEE11iAuthenticationMode' -> 'PSKAuthentication',
          'WPAEncryptionModes' -> 'TKIPEncryption',
          'WPAAuthenticationMode' -> 'PSKAuthentication',
          'BasicEncryptionModes' -> 'WEPEncryption',
          'BasicAuthenticationMode' -> 'SharedAuthentication'
      ).get(key.keyName)
    )"""

    isWpa_Wpa2 = """if(
      realValue('BeaconType') == 'WPAand11i' and
      realValue('WPAEncryptionModes') == 'TKIPEncryption' and
      realValue('WPAAuthenticationMode') == 'PSKAuthentication' and
      realValue('IEEE11iEncryptionModes') == 'AESEncryption' and
      realValue('IEEE11iAuthenticationMode') == 'PSKAuthentication'
    ) 'WPA_WPA2'"""

    setWpa_Wpa2 = """if(value == 'WPA_WPA2')(
      map('BeaconType' -> 'WPAand11i',
          'WPAEncryptionModes' -> 'TKIPEncryption',
          'WPAAuthenticationMode' -> 'PSKAuthentication',
          'IEEE11iEncryptionModes' -> 'AESEncryption',
          'IEEE11iAuthenticationMode' -> 'PSKAuthentication',
          'BasicEncryptionModes' -> 'None',
          'BasicAuthenticationMode' -> 'None'
      ).get(key.keyName)
    )"""
  }
