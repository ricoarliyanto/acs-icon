    # Convert Manufacturer value to uppercase
    Manufacturer {
      expression = """${singleValue.toUpperCase()}"""
      type = expression
    }

    WAN.Internet {
      Status = ${tools.expression} {
        expression = """${if(singleValue.contains('Up')) 'Connected'
        else if(set('Down', 'Disabled').exists(singleValue.contains(_))) 'Disconnected'

      Mac = ${tools.readConverters.oneToOne.stripMAC}
