remappingStrategy.wifiSecurity {

  isNone = """if(
      realValue('BeaconType') == 'Basic' and
      realValue('BasicEncryptionModes') == 'None'
    ) 'NONE'"""

  setNone = """if(value == 'NONE')(
      map('BeaconType' -> 'Basic',
          'BasicEncryptionModes' -> 'None'
      ).get(key.keyName)
    )"""

  isWep = """if(
      realValue('BeaconType') == 'Basic' and
      realValue('BasicEncryptionModes') == 'WEPEncryption'
    ) 'WEP'"""

  setWep = """if(value == 'WEP')(
      map('BeaconType' -> 'Basic',
          'BasicEncryptionModes' -> 'WEPEncryption'
      ).get(key.keyName)
    )"""
}

realPatching.patching {
  # this device when its wifi is disabled returns empty PossibleChannels parameter, in such situations we will put
  # placeholder values in appropriate VMD keys (mostly so the channel ranking mechanisms doesn't blow up in our faces)

  root.LANDevice.1.WLANConfiguration.65094.PossibleChannels {
    %patch-value = "${if(node.value == '') range(1,13).map(x => x.toString).join(',') else node.value.toString}"
  }
}
