remappingStrategy {
  read_converter {
    LAN.Devices.%i {
      MACAddress = ${tools.readConverters.oneToOne.stripMAC}
      InterfaceType = ${tools.expression} {
        expression = """${
          if(realValue('InterfaceType').contains('Wi-Fi') or realValue('PhysAddress').stripMac == '3C52A156447A')
            '802.11'
          else if(realValue('InterfaceType').contains('Ethernet'))
            'Ethernet'
          else
            'Unknown'}"""
      }
      AddressSource = ${tools.expression} {
        expression = """${if (singleValue == '')
            'None'
          else 'DHCP'}"""
      }
    }
  }
  mapping {
    root.X_AVS_EXT.LAN {
      DevicesCount.real = "root.Hosts.HostNumberOfEntries"
      Devices.realObject = "root.Hosts.Host"
      Devices.%i.realObject = "root.Hosts.Host.%i"
      Devices.%i {
        Active.real = "root.Hosts.Host.%i.Active"
        AddressSource.real = "root.Hosts.Host.%i.DHCPClient"
        HostName.real = "root.Hosts.Host.%i.HostName"
        IPAddress.real = "root.Hosts.Host.%i.IPAddress"
        InterfaceType.real = ["root.Hosts.Host.%i.InterfaceType","root.Hosts.Host.%i.PhysAddress"]
        LeaseTimeRemaining.real = "root.Hosts.Host.%i.LeaseTimeRemaining"
        MACAddress.real = "root.Hosts.Host.%i.PhysAddress"
        Layer1Interface.real = "root.Hosts.Host.%i.Layer1Interface"
        Layer3Interface.real = "root.Hosts.Host.%i.Layer3Interface"
      }
    }
  }
