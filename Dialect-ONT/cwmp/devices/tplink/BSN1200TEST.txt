virtualPatching.patching {
root.X_AVS_EXT {
  WiFi {
    %any {
      SupportedStandards.%force-value = "a,n,ac"
    }
    
    1{
      AssociatedDevice.%foreach-i {
        %in = """${dm.LANDevice.`1`.Hosts.Host.any.keys.map(k=>k.keyName).filter(keyName=>
          dm.LANDevice.`1`.Hosts.Host.selectDynamic(keyName).selectDynamic("Layer2Interface").value.useAs(layer2InterfaceValue=>
          if(layer2InterfaceValue.take(1).equalsIgnoreCase("w") and layer2InterfaceValue.substring(2,3).equalsIgnoreCase("2")) false
          else true))}"""
        %drop = true
      }
    }
    
    5{
      AssociatedDevice.%foreach-i {
        %in = """${dm.LANDevice.`1`.Hosts.Host.any.keys.map(k=>k.keyName).filter(keyName=>
          dm.LANDevice.`1`.Hosts.Host.selectDynamic(keyName).selectDynamic("Layer2Interface").value.useAs(layer2InterfaceValue=>
          if(layer2InterfaceValue.take(1).equalsIgnoreCase("w") and layer2InterfaceValue.substring(2,3).equalsIgnoreCase("5")) false
          else true))}"""
        %drop = true
      }
    }
   
    
    
    %force-object-exists = true
    1 {
      %force-object-exists = true
      AutoChannelEnable.%force-value = true
      Band.%force-value = "2.4"
      PossibleChannels.%force-value = "1,2,3,4,5,6,7,8,9"
      Status.%force-value = "up"
      SupportedChannelBandwidth.%force-value = "20MHz,40MHz,Auto"
    }
    5 {
      Band.%force-value = "5"
      AutoChannelEnable.%force-value = true
      SupportedChannelBandwidth.%force-value = "20MHz,40MHz,80MHz,160MHz,Auto"
    }
    6{
      %drop=true
    }
    7{
      %drop=true
    }
    8{
      %drop=true
    }
  }
}

root.X_AVS_EXT.Diagnostics.Neighborhood.Run = ${tools.patching.executable}
}
