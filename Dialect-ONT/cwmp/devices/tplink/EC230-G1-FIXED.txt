    root.X_AVS_EXT {
        WAN {
          UpTime.real = "${sv.vdmWan1.parseJson.interfaceKey.asString.toParameterKey}.Uptime"

          Downstream {

            Packets.real = "root.Ethernet.VLANTermination.1.Stats.PacketsReceived"

          Upstream {

          }
        }

            Bytes.real = "root.Ethernet.VLANTermination.1.Stats.BytesSent"

   WAN.UpTime.real = "${sv.vdmWan1.parseJson.interfaceKey.asString.toParameterKey}.Uptime"
