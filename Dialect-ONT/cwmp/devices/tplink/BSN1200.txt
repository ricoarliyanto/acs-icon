remappingStrategy {
  activeKey.wifiScan1 = "root.X_TP_NeighboringWiFiDiagnostic"
  
   read_converter {
    LAN.Devices.%i {
      InterfaceType = ${tools.expression} {
        expression = """${
          if(singleValue.contains('WL'))
            '802.11'
          else if(singleValue.contains('LAN'))
            'Ethernet'
          else
            'Unknown'}"""
      }
    }
  }
  
    mapping {
    root.X_AVS_EXT {
      WiFi {
        1 {
          Enable.real = "root.LANDevice.1.WLANConfiguration.1.Enable"
          OperatingChannelBandwidth.real = "root.LANDevice.1.WLANConfiguration.1.X_TP_Bandwidth"
        }
        5 {
          Enable.real = "root.LANDevice.1.WLANConfiguration.2.Enable"
          OperatingChannelBandwidth.real = "root.LANDevice.1.WLANConfiguration.2.X_TP_Bandwidth"
        }
      }
      WAN.Stats {
        Downstream {
          Bytes.real = "root.WANDevice.1.WANEthernetInterfaceConfig.Stats.BytesReceived"
          Packets.real = "root.WANDevice.1.WANEthernetInterfaceConfig.Stats.PacketsReceived"
        }
        Upstream {
          Bytes.real = "root.WANDevice.1.WANEthernetInterfaceConfig.Stats.BytesSent"
          Packets.real = "root.WANDevice.1.WANEthernetInterfaceConfig.Stats.PacketsSent"
        }
      }
      LAN {
        Devices.%i {
        InterfaceType.real = "root.LANDevice.1.Hosts.Host.%i.Layer2Interface"
        }
      }
    }
  }
}



virtualPatching.patching {
  root.X_AVS_EXT {
    WiFi {
      %any {
        SupportedStandards.%force-value = "a,n,ac"
      }
      1 {
        SupportedChannelBandwidth.%force-value = "20MHz,40MHz,Auto"
      }
      5 {
        SupportedChannelBandwidth.%force-value = "20MHz,40MHz,80MHz,160MHz,Auto"
      }
    }
  }
}
