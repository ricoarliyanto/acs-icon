remappingStrategy {
  read_converter {
    LAN {
      Devices.%i {
        InterfaceType = ${tools.expression} {
          expression = """${
            if(singleValue.contains('Wi-Fi'))
              '802.11'
            else if(singleValue.contains('Other'))
              'Ethernet'
            else
              'Unknown' }"""
      }
    }
  }
}
  mapping {
    root.X_AVS_EXT {
      LAN {
        Devices.%i {
          InterfaceType.real = "root.Hosts.Host.%i.InterfaceType"
        }
      }
    }
  }
}

virtualPatching.patching {
  root.X_AVS_EXT {
    WiFi {
      %any {
        SupportedStandards = null
        SupportedStandards = {}
      }
    }
  }
}
